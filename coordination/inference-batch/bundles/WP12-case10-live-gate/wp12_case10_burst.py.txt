"""WP-12 case-10 live gate: N-wide burst on one role via the placement-queue
transport (call_orchestrator_forced, request_priority=background,
workload_class=eval_batch, force_mode=direct, allow_delegation=False).

Evidence emitted (single JSON to stdout):
  - per-request: ok, latency_s, session_id, n chars
  - slots_timeline: per-poll busy vector across the fleet's quarter ports
    (direct llama-server /slots — bypasses the API rate limiter)
  - max_concurrent_busy: peak simultaneously-busy quarters
  - ports_served (from tap events tail): distinct ports + role/topology_role
    per request  -> single-CAB fleet-identity fact + no-phantom-full fact
  - contention_before/after: /dashboard/api/contention migration counters +
    per_role_scheduling diff -> zero-migration / zero-fallback-churn fact

Operator inference grant: 2026-07-23 "inference granted for case-10 - run the
live gate". Observation-only (MEASUREMENT.md): facts gate the WP-12 acceptance
checkbox; no performance number here gates anything.
"""
import argparse
import json
import sys
import threading
import time
from pathlib import Path

ORCH = Path("/mnt/raid0/llm/epyc-orchestrator")
sys.path.insert(0, str(ORCH / "scripts" / "benchmark"))
from seeding_orchestrator import call_orchestrator_forced  # noqa: E402

import httpx  # noqa: E402

TAP = Path("/mnt/raid0/llm/tmp/inference_tap_events.jsonl")


def snapshot_contention(url: str) -> dict:
    try:
        r = httpx.get(f"{url}/dashboard/api/contention", timeout=10)
        d = r.json()
        return {
            "migration_counters": d.get("migration_counters"),
            "per_role_scheduling": d.get("per_role_scheduling"),
        }
    except Exception as exc:  # noqa: BLE001
        return {"error": str(exc)}


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--role", required=True)
    ap.add_argument("--ports", required=True, help="comma list of fleet quarter ports")
    ap.add_argument("--n", type=int, default=4)
    ap.add_argument("--max-tokens", type=int, default=300)
    ap.add_argument("--url", default="http://localhost:8000")
    ap.add_argument("--timeout", type=int, default=240)
    args = ap.parse_args()
    ports = [int(p) for p in args.ports.split(",")]

    tap_offset = TAP.stat().st_size if TAP.exists() else 0
    before = snapshot_contention(args.url)

    prompt = ("List the integers from 1 to 250, one per line, "
              "with no commentary and no other text.")
    results: list[dict] = [None] * args.n  # type: ignore[list-item]
    start_barrier = threading.Barrier(args.n + 1)

    def fire(i: int) -> None:
        sid = f"wp12-case10-{args.role}-burst-{i}"
        client = httpx.Client(timeout=args.timeout)
        start_barrier.wait()
        t0 = time.monotonic()
        try:
            resp = call_orchestrator_forced(
                prompt=prompt,
                force_role=args.role,
                force_mode="direct",
                url=args.url,
                timeout=args.timeout,
                client=client,
                allow_delegation=False,
                session_id=sid,
                request_priority="background",
                workload_class="eval_batch",
                max_tokens=args.max_tokens,
            )
            txt = (resp or {}).get("response") or (resp or {}).get("content") or ""
            results[i] = {
                "session_id": sid,
                "ok": bool(resp) and not (resp or {}).get("error"),
                "latency_s": round(time.monotonic() - t0, 2),
                "chars": len(str(txt)),
                "error": (resp or {}).get("error"),
            }
        except Exception as exc:  # noqa: BLE001
            results[i] = {
                "session_id": sid,
                "ok": False,
                "latency_s": round(time.monotonic() - t0, 2),
                "error": f"{type(exc).__name__}: {exc}",
            }
        finally:
            client.close()

    threads = [threading.Thread(target=fire, args=(i,)) for i in range(args.n)]
    for t in threads:
        t.start()

    timeline: list[dict] = []
    slots_client = httpx.Client(timeout=3)
    start_barrier.wait()  # release the burst
    t_burst0 = time.monotonic()
    while any(t.is_alive() for t in threads):
        busy = {}
        for p in ports:
            try:
                slots = slots_client.get(f"http://localhost:{p}/slots").json()
                busy[str(p)] = any(s.get("is_processing") for s in slots)
            except Exception:  # noqa: BLE001
                busy[str(p)] = None
        timeline.append({"t": round(time.monotonic() - t_burst0, 2), "busy": busy})
        time.sleep(0.3)
    for t in threads:
        t.join()
    slots_client.close()

    after = snapshot_contention(args.url)

    tap_events = []
    if TAP.exists():
        with TAP.open("rb") as fh:
            fh.seek(tap_offset)
            for line in fh.read().decode(errors="replace").splitlines():
                try:
                    e = json.loads(line)
                except Exception:  # noqa: BLE001
                    continue
                tap_events.append({k: e.get(k) for k in (
                    "ts", "role", "topology_role", "port", "session_id")})

    max_busy = max(
        (sum(1 for v in row["busy"].values() if v) for row in timeline),
        default=0,
    )
    burst_ports = sorted({
        e["port"] for e in tap_events
        if e.get("role") == args.role and e.get("port") is not None
    })
    print(json.dumps({
        "kind": "wp12_case10_burst",
        "role": args.role,
        "fleet_ports": ports,
        "n": args.n,
        "requests": results,
        "max_concurrent_busy": max_busy,
        "slots_timeline": timeline,
        "tap_events_burst": tap_events,
        "distinct_ports_served": burst_ports,
        "contention_before": before,
        "contention_after": after,
    }, indent=1, default=str))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
